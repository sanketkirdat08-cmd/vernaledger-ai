@app.post("/api/ocr")
async def ocr_receipt(req: OCRRequest):
    # 1. Base64 Image Decode & Lanczos sharpening
    img_bytes = base64.b64decode(req.image_base64)
    img = Image.open(io.BytesIO(img_bytes)).convert("RGB")
    
    # 2. Call Google Gemini on TPUs via REST for sub-2s speed
    raw_json = call_gemini_multimodal_rest(api_key, "gemini-flash-lite-latest", PROMPT, img)
    items = json.loads(raw_json)

    # 3. Smart Kirana Rate vs. Total Disambiguation Algorithm
    for it in items:
        qty = float(it.get("qty") or 1.0)
        rate = float(it.get("rate") or 0.0)
        total = float(it.get("total") or 0.0)
        if rate == 0.0 and total > 0.0:
            it["rate"] = round(total / qty, 2)  # e.g., 10kg ₹500 -> ₹50/kg
        elif total == 0.0 and rate > 0.0:
            it["total"] = round(rate * qty, 2)
    return {"success": True, "items": items}

# 2. Server-Side Regional Kirana NLP Token Parsing (api/index.py)
def parse_voice_billing_nlp(transcript: str, today_str: str) -> dict:
    # Matches: "रमेशभाऊ २ किलो साखर १०० रुपये"
    customer_match = re.search(r'([A-Za-z\u0900-\u097F]+)(?:भाऊ|शेठ|काका|राव)', transcript)
    customer_name = customer_match.group(0) if customer_match else "रोख ग्राहक"
    
    # Extract numerical quantities, regional units, and items
    item_matches = re.findall(r'(\d+(?:\.\d+)?)\s*(किलो|kg|ग्रॅम|लीटर|लिटर|नग)\s*([A-Za-z\u0900-\u097F]+)\s*(\d+)?', transcript)
    parsed_items = [{"name": m[2], "qty": float(m[0]), "unit": m[1], "rate": float(m[3] or 0)} for m in item_matches]
    return {"customer_name": customer_name, "items": parsed_items}

# 1. Real-Time Atomic Stock Deduction Engine (api/index.py)
def deduct_inventory_stock(items: list, conn) -> list:
    cursor = conn.cursor()
    for it in items:
        name = it.get("name", "").strip()
        billed_qty = float(it.get("qty") or 0.0)
        # Atomic update clamped at 0 to prevent negative inventory
        sql = "UPDATE shop_inventory SET stock_qty = MAX(0.0, stock_qty - ?) WHERE LOWER(item_name) = LOWER(?)"
        cursor.execute(sql, (billed_qty, name))
    conn.commit()
